# 📊 Crisp Pulse 知识脉冲看板

## 📈 核心指标

```dataviewjs
const allPages = dv.pages('!"templates"');
const now = dv.luxon.DateTime.now();
const weekAgo = now.minus({days: 7});

const totalNotes = allPages.length;
const weekNotes = allPages.where(p => p.file.ctime >= weekAgo).length;
const recentEdited = allPages.where(p => p.file.mtime >= weekAgo).length;

let totalWords = 0;
allPages.forEach(p => { if (p.file.size) totalWords += Math.round(p.file.size / 4); });

const tags = new Set();
allPages.forEach(p => { (p.file.tags || []).forEach(t => tags.add(t)); });

const cards = [
  { label: "总笔记数", value: totalNotes, unit: "篇" },
  { label: "本周新建", value: weekNotes, unit: "篇" },
  { label: "本周编辑", value: recentEdited, unit: "篇" },
  { label: "总字数（估）", value: (totalWords/1000).toFixed(1), unit: "k" },
  { label: "标签数", value: tags.size, unit: "个" },
];

let html = '<div class="crisp-stats-grid">';
cards.forEach(c => {
  html += `<div class="crisp-stat-card">
    <div class="crisp-stat-value">${c.value}<span style="font-size:14px;font-weight:400;color:#999"> ${c.unit}</span></div>
    <div class="crisp-stat-label">${c.label}</div>
  </div>`;
});
html += '</div>';

dv.container.innerHTML = html;
```

## 🔥 活跃热力图

> [!note] 年度知识活跃脉冲
> ```dataviewjs
> const calendarData = {
>   year: 2026,
>   colors: {
>     green: ["#ebedf0", "#9be9a8", "#40c463", "#30a14e", "#216e39"]
>   },
>   entries: []
> };
>
> for (let page of dv.pages()) {
>   calendarData.entries.push({
>     date: page.file.ctime.toFormat("yyyy-MM-dd"),
>     intensity: 1
>   });
> }
>
> renderHeatmapCalendar(this.container, calendarData);
> ```

## 📅 最近笔记

> [!note] 最近 10 天编辑的笔记
> ```dataview
> TABLE file.mtime AS "最后编辑", file.tags AS "标签"
> WHERE file.cday >= date(today) - dur(10 days)
> SORT file.mtime DESC
> LIMIT 10
> ```
