## 五、API 封装（含字典表支持）

### `frontend/src/api/analysis.ts`

```typescript
import { get, post, put } from '../utils/request'

// ============ APIs ============

// 上传（仅 preschool_budget_admin）
export function uploadFiles(period: string, files: File[]) {
  const formData = new FormData()
  formData.append('period', period)
  files.forEach(f => formData.append('files', f))
  return post<{ uploadId: number }>('/analysis/upload', formData, {
    headers: { 'Content-Type': 'multipart/form-data' },
  })
}

// 园所列表（preschool_budget_admin, boss, operation_director, region_finance）
export function getGardens(params: { uploadId?: number; keyword?: string; status?: 'active' | 'closed'; page: number; size: number }) {
  return get<{ list: GardenSummary[]; total: number }>('/analysis/gardens', params)
}

// 单园看板（以上 + principal）
export function getGardenDashboard(code: string, uploadId?: number) {
  return get<DashboardData>(`/analysis/garden/${code}`, { uploadId })
}

// 学前经营看板（仅 preschool_budget_admin, boss, operation_director）
export function getPreschoolDashboard(uploadId?: number) {
  return get<GroupDashboardData>('/analysis/preschool-dashboard', { uploadId })
}

// 学前预算分析表（preschool_budget_admin, boss, operation_director, region_finance, principal）
export function getBudgetAnalysis(params: { uploadId?: number; scope?: string }) {
  return get('/analysis/budget-analysis', params)
}

// 科目字典（除 group_admin 外全部）
export function getSubjectDict() {
  return get<SubjectDictItem[]>('/analysis/subject-dict')
}

// 园长映射（仅 preschool_budget_admin）
export function getPrincipalMappings() { return get('/analysis/principal-mappings') }
export function updatePrincipalMapping(principalId: number, companyIds: number[]) {
  return put(`/analysis/principal-mappings/${principalId}`, { companyIds })
}
```

## 八、ECharts 集成方案

需要额外注册瀑布图所需的 bar chart：

```bash
npm install echarts vue-echarts
```

```typescript
// main.ts
import ECharts from 'vue-echarts'
import { use } from 'echarts/core'
import { PieChart, BarChart, ScatterChart, LineChart } from 'echarts/charts'
import { GridComponent, TooltipComponent, LegendComponent, TitleComponent } from 'echarts/components'
import { CanvasRenderer } from 'echarts/renderers'

use([CanvasRenderer, PieChart, BarChart, ScatterChart, LineChart, GridComponent, TooltipComponent, LegendComponent, TitleComponent])
```
