```python
# 3. 登录 + 上传数据（在 Python 环境执行）
import requests, os

BASE = 'http://localhost:8094'
# 登录
resp = requests.post(f'{BASE}/api/v1/auth/dev-login', json={'employeeNo': 'BUDGET01'})
TOKEN = resp.json()['data']['token']

# 上传
src = r'D:\财务系统\doc\阳阳给的出纳表0617'
files = [
    '01.学前教育-全年预算[财年].xlsx',
    '02.学前教育-期间预算[期间] .xlsx',
    '2025年1-4月实际报表.xlsx',
    '2025年4月实际报表.xlsx',
    '2026年1-4月实际报表.xlsx',
    '2026年4月实际报表.xlsx',
]
form_files = []
for f in files:
    path = os.path.join(src, f)
    if os.path.exists(path):
        form_files.append(('files', (f, open(path, 'rb'), 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet')))

resp = requests.post(f'{BASE}/api/v1/analysis/upload',
    headers={'Authorization': f'Bearer {TOKEN}'},
    data={'period': '2026-04'}, files=form_files, timeout=120)
print(resp.json())
```

⚠️ **不能用 curl + bash**：文件名里 `[财年]` 会被 bash 当 glob 吞掉。
